Explanation of Changes                                                          

 1 Directory Setup: Creates the target installation directory /app/sqlite and   
   temporary build directory /tmp/sqlite-build.                                 
 2 Extraction: Unpacks /app/vendor/sqlite-fossil-release.tar.gz into the build  
   directory.                                                                   
 3 Compilation with gcov: Passes CFLAGS="-g -O0 -fprofile-arcs -ftest-coverage" 
   and LDFLAGS="-fprofile-arcs -ftest-coverage" to ./configure and              
   builds/installs SQLite using make and make install.                          
 4 PATH Integration: Configures /etc/profile.d/sqlite.sh and appends export

setup.sh                                                                        

                                                                                
#!/bin/bash                                                                     
                                                                                
set -euo pipefail                                                               
                                                                                
                                                                                
                                                                                
BUILD_DIR="/tmp/sqlite-build"                                                   
                                                                                
INSTALL_DIR="/app/sqlite"                                                       
                                                                                
TARBALL="/app/vendor/sqlite-fossil-release.tar.gz"                              
                                                                                
                                                                                
                                                                                
echo "=== Preparing build environment ==="                                      
                                                                                
rm -rf "$BUILD_DIR"                                                             
                                                                                
mkdir -p "$BUILD_DIR"                                                           
                                                                                
mkdir -p "$INSTALL_DIR"                                                         
                                                                                
                                                                                
                                                                                
echo "=== Extracting SQLite source ==="                                         
                                                                                
tar -xzf "$TARBALL" -C "$BUILD_DIR"                                             
                                                                                
                                                                                
                                                                                
SRC_DIR=$(find "$BUILD_DIR" -maxdepth 2 -name "configure" -exec dirname {} \; | 
head -n 1)                                                                      
                                                                                
if [ -z "$SRC_DIR" ]; then                                                      
                                                                                
    echo "Error: configure script not found in extracted archive." >&2          
                                                                                
    exit 1                                                                      
                                                                                
fi                                                                              
                                                                                
                                                                                
                                                                                
cd "$SRC_DIR"                                                                   
                                                                                
                                                                                
                                                                                
echo "=== Configuring SQLite with gcov instrumentation ==="                     
                                                                                
export CFLAGS="-g -O0 -fprofile-arcs -ftest-coverage"                           
                                                                                
export LDFLAGS="-fprofile-arcs -ftest-coverage"                                 
                                                                                
                                                                                
                                                                                
./configure --prefix="$INSTALL_DIR" CFLAGS="$CFLAGS" LDFLAGS="$LDFLAGS"         
                                                                                
                                                                                
                                                                                
echo "=== Building SQLite ==="                                                  
                                                                                
NPROC=$(nproc 2>/dev/null || echo 2)                                            
                                                                                
make -j"$NPROC"                                                                 
                                                                                
                                                                                
                                                                                
echo "=== Installing SQLite ==="                                                
                                                                                
make install                                                                    
                                                                                
                                                                                
                                                                                
# Create symlink at /app/sqlite/sqlite3 if not directly installed there         
                                                                                
if [ -f "$INSTALL_DIR/bin/sqlite3" ] && [ ! -f "$INSTALL_DIR/sqlite3" ]; then   
                                                                                
    ln -sf "$INSTALL_DIR/bin/sqlite3" "$INSTALL_DIR/sqlite3"                    
                                                                                
fi                                                                              
                                                                                
                                                                                
                                                                                
echo "=== Updating PATH ==="                                                    
                                                                                
PROFILE_SCRIPT="/etc/profile.d/sqlite.sh"                                       
                                                                                
if [ -d /etc/profile.d ]; then                                                  
                                                                                
    echo "export PATH=\"$INSTALL_DIR/bin:\$PATH\"" | tee "$PROFILE_SCRIPT" >    
/dev/null                                                                       
                                                                                
    chmod +x "$PROFILE_SCRIPT"                                                  
                                                                                
fi                                                                              
                                                                                
                                                                                
                                                                                
for rcfile in ~/.bashrc ~/.profile /etc/bash.bashrc; do                         
                                                                                
    if [ -f "$rcfile" ]; then                                                   
                                                                                
        if ! grep -q "$INSTALL_DIR/bin" "$rcfile"; then                         
                                                                                
            echo "export PATH=\"$INSTALL_DIR/bin:\$PATH\"" >> "$rcfile"         
                                                                                
        fi                                                                      
                                                                                
    fi                                                                          
                                                                                
done                                                                            
                                                                                
                                                                                
                                                                                
# Cleanup temporary build files                                                 
                                                                                
cd /app                                                                         
                                                                                
rm -rf "$BUILD_DIR"                                                             
                                                                                
                                                                                
                                                                                
echo "=== SQLite installation complete ==="                                     
                                                                                
"$INSTALL_DIR/bin/sqlite3" --version
